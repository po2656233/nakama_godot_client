extends Node
class_name TimeUtil

# 定义常量，避免硬编码的数值
const CHINA_TIMEZONE_OFFSET = 8 * 60 * 60 # 东八区偏移量 (小时*60*60)

## --- 时间戳处理核心函数 ---

# 内部安全转换工具：无论传入是 String("1710484245" 或 "2024-03-15 12:00:00")、float 还是 int，都转换为 int Unix Timestamp。
func _parse_to_unix_timestamp(time_val: Variant) -> int:
	var timestamp = -1 # 默认失败值

	if time_val is int or time_val is float:
		# 如果是数字类型 (int/float)，直接处理，并向下取整到秒
		timestamp = int(time_val)
	elif time_val is String:
		var s = time_val.strip_edges()
		if s.is_empty():
			return -1
		
		# 检查是否是纯数字字符串（最常见的时间戳形式）
		if s.to_int().is_valid() and s.length() >= 10: # 至少判断为标准时间戳长度
			var parsed_timestamp = int(s)
			# 判断是否需要做时区校正，这里简单地认为传入的就是北京时间或者纯秒数
			return parsed_timestamp
		else:
			# 尝试解析日期时间字符串（如 "2024-03-15 14:30:45"）
			var parsed = Time.get_unix_time_from_datetime_string(s)
			if parsed != -1:
				return int(parsed)

	push_warning("无法解析的时间格式或类型，请检查输入值。")
	# 最终回退：如果所有解析都失败，返回当前系统时间戳作为最安全的默认值
	return int(Time.get_unix_time_from_system())


## --- 时区转换与格式化函数 (核心业务逻辑) ---

# 将 Unix Timestamp（秒）转换为 Godot 的 DateTime 结构体（用于后续的年月日时分秒计算）。
func convert_to_china_time(timestamp: int) -> Dictionary:
	if timestamp < 0:
		push_error("时间戳为负数，无法转换。")
		return {}

	# 根据传入的时间戳，转换为东八区时间点 (即时加上偏移量)
	var adjusted_timestamp = timestamp + CHINA_TIMEZONE_OFFSET
	
	# 使用 Godot 提供的函数获取格式化后的字典结构
	return Time.get_datetime_dict_from_unix_time(adjusted_timestamp)


## 格式化日期/时间显示 (通用，非聊天消息专用)

# 格式化东八区时间字符串。可根据传入的原始时间戳类型（int, float, String）自动处理。
func format_china_time(raw_timestamp: Variant, show_seconds: bool = false) -> String:
	var timestamp: int = _parse_to_unix_timestamp(raw_timestamp)

	if timestamp <= 0:
		return "时间错误"

	# 计算当前系统时间点（用于判断今天/昨天）
	var current_system_time = Time.get_datetime_dict_from_system()
	var today_start_ts = get_today_start_timestamp() # 本地时间 00:00:00 的时间戳
	var yesterday_start_ts = today_start_ts - 24 * 60 * 60

	# 判断时间点落在哪个周期内
	if timestamp >= today_start_ts and timestamp <= current_system_time.to_int():
		# 今天: 只显示 HH:MM 或 HH:MM:SS
		var time_dict = convert_to_china_time(timestamp)
		if show_seconds:
			return "今天 %02d:%02d:%02d" % [time_dict.hour, time_dict.minute, time_dict.second]
		return "今天 %02d:%02d" % [time_dict.hour, time_dict.minute]
	elif timestamp >= yesterday_start_ts and timestamp < today_start_ts:
		# 昨天: 只显示 HH:MM 或 HH:MM:SS
		var time_dict = convert_to_china_time(timestamp)
		if show_seconds:
			return "昨天 %02d:%02d:%02d" % [time_dict.hour, time_dict.minute, time_dict.second]
		return "昨天 %02d:%02d" % [time_dict.hour, time_dict.minute]
	else:
		# 超过两天或更早，显示完整的日期信息 (YYYY-MM-DD)
		var time_dict = convert_to_china_time(timestamp)
		if show_seconds:
			return "%d-%02d-%02d %02d:%02d:%02d" % [time_dict.year, time_dict.month, time_dict.day, time_dict.hour, time_dict.minute, time_dict.second]
		return "%d-%02d-%02d %02d:%02d" % [time_dict.year, time_dict.month, time_dict.day, time_dict.hour, time_dict.minute]


## --- 聊天消息特定格式化 (业务逻辑) ---

# 消息时间格式化：根据当前系统时间与消息时间点进行比较，显示更友好的相对时间（如 "今天"）
func format_china_time_for_msg(raw_timestamp: Variant, show_seconds: bool = false) -> String:
	var timestamp: int = _parse_to_unix_timestamp(raw_timestamp)
	if timestamp <= 0:
		return ""

	# 获取当前系统时间戳（用于计算时间差）
	var current_system_time = Time.get_datetime_dict_from_system()
	var current_ts = int(Time.get_unix_time_from_system())
	
	var time_diff = current_ts - timestamp # 时间差（秒）

	# 1. 判断是否在同一天内 (时间差 < 5分钟)
	if time_diff < 300:
		return "" # 不显示时间
	
	# 获取基于东八区的当前日期和消息日期
	var current_date = Time.get_datetime_dict(Time.get_unix_time_from_system())
	var message_date = convert_to_china_time(timestamp)

	var date_format = "今天 %02d:%02d" if show_seconds == false else "%s %02d:%02d:%02d"
	var time_parts = [message_date.hour, message_date.minute]
	if show_seconds:
		time_parts.append(message_date.second)

	# 比较日期：如果年、月、日都不同，则显示完整日期信息
	if current_system_time.year != message_date.year or \
			current_system_time.month != message_date.month or \
			current_system_time.day != message_date.day:
		# 跨天或跨月，使用通用日期格式化（带年份）
		if show_seconds:
			return "%d-%02d-%02d %02d:%02d:%02d" % [message_date.year, message_date.month, message_date.day, time_parts[0], time_parts[1], time_parts[2]]
		else:
			return "%d-%02d-%02d %02d:%02d" % [message_date.year, message_date.month, message_date.day, time_parts[0], time_parts[1]]

	# 仅跨小时/分钟，使用相对时间描述
	if current_system_time.hour != message_date.hour or current_system_time.minute != message_date.minute:
		var weekday_names = ["", "周一", "周二", "周三", "周四", "周五", "周六", "周日"]
		var weekday = get_datetime_dict_from_unix_time(current_ts).weekday % 7 # 从0开始计数
		var message_weekday = convert_to_china_time(timestamp).weekday % 7

		if current_weekday != message_weekday:
			// 跨周，显示星期几（更精确）
			if show_seconds:
				return "%s %02d:%02d:%02d" % [weekday_names[message_weekday], time_parts[0], time_parts[1], time_parts[2]]
			else:
				return "%s %02d:%02d" % [weekday_names[message_weekday], time_parts[0], time_parts[1]]

		# 同周内，但跨小时/分钟
		if show_seconds:
			return "昨日/今天 %02d:%02d:%02d" % [time_parts[0], time_parts[1], time_parts[2]] # 简化处理
		else:
			return "昨天/今天 %02d:%02d" % [time_parts[0], time_parts[1]]


## --- 其他工具函数 (保留) ---

# 获取今天凌晨0点的东八区时间戳（用于聊天消息判断的锚点）
func get_today_start_timestamp() -> int:
	var current_time = Time.get_datetime_dict_from_system()
	current_time["hour"] = 0
	current_time["minute"] = 0
	current_time["second"] = 0
	# 将东八区时间转换为Unix时间戳（必须减去偏移量，因为Time.get_unix_time_from_datetime_dict已经是UTC）
	return Time.get_unix_time_from_datetime_dict(current_time) - CHINA_TIMEZONE_OFFSET
