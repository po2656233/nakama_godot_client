## --- 网络调用封装 (替代原 util.gd 的 Asyn_Call) ---

# 执行一个异步网络/服务调用，提供重试和回调机制。
# @param caller: 调用此方法的节点对象（用于日志记录）。
# @param method_name: 要调用的目标方法名。
# @param args: 传递给目标方法的参数数组。
# @param on_success: 成功时的回调函数 (Callable)。
# @param on_failure: 失败或重试次数耗尽时的回调函数 (Callable)。
# @param inqueue: 是否应该将调用添加到可靠消息队列进行自动重试。
func execute_async_call(caller: Object, method_name: String, args: Array, on_success: Callable, on_failure: Callable, inqueue: bool = true) -> bool:
	print("NetworkWrapper::execute_async_call - 开始异步调用...")

	# 1. 前置校验：检查方法是否存在
	if not caller or not caller.has_method(method_name):
		push_error("目标对象 %s 不存在或没有名为 '%s' 的方法。" % [str(caller), method_name])
		on_failure.call("MethodNotFound") if on_failure and on_failure.is_valid() else return false

	# 2. 执行实际调用并等待结果 (使用 await 实现协程同步，这是Godot 4的最佳实践)
	var task = caller.callv(method_name, args)
	if not (task is Object) or !task.has_method("is_valid") or !task.is_valid():
		push_error("无法发起异步任务：调用方法返回了无效对象。")
		on_failure.call("InvalidAsyncTask") if on_failure and on_failure.is_valid() else return false

	var result = await task.completed
	
	# 3. 结果处理逻辑
	if result != null and result.has_method("is_exception") and result.is_exception():
		var errMsg = result.get_exception().message
		push_error("异步调用捕获到异常: %s" % errMsg)

		# 判断是否是网络相关的、可重试的错误（需要与上游协议保持一致）
		if errMsg in ["AuthenticationFailed", "TimeoutException"]: # 示例：匹配业务级网络错误码
			var final_error = "NetworkFailure: " + errMsg
			on_failure.call(final_error) if on_failure and on_failure.is_valid() else return false

			if inqueue:
				# 将失败的调用封装到消息队列进行重试管理
				var msg_id = _reliable_queue.EnqueueMessage(method_name, args, caller, on_success, on_failure)
				if not msg_id:
					print("WARN: 无法将网络错误添加到消息队列进行重试。")

		# 其他非可重试的错误直接报告失败
		else:
			on_failure.call(errMsg) if on_failure and on_failure.is_valid() else return false
		return false # 视为调用流程失败，但已通知用户/系统

	# 4. 成功处理
	print("NetworkWrapper::execute_async_call - 成功接收结果。")
	if on_success and on_success.is_valid():
		on_success.call(result) # 执行成功回调
	return true
