# MatchService.gd
# Handles Nakama match creation, joining, and matchmaking.
extends Node

# Dependency on NakamaCore (assumed to be in Global)
var core: Node

# Configuration
@export var min_players: int = 2
@export var max_players: int = 4
@export var client_version: String = "dev"

# State
var match_id: String = ""
var matchmaker_ticket: String = ""

# Signals
signal match_created(m_id: String)
signal match_joined(m_id: String)
signal matchmaker_matched(players: Array)
signal match_error(message: String)
signal match_disconnected()

func _ready() -> void:
	if "NakamaCore" in Global:
		core = Global.NakamaCore
	else:
		push_error("[MatchService] NakamaCore not found in Global!")
		return
	
	# Connect to core socket events
	core.socket_event_received.connect(_on_socket_event)

func _on_socket_event(event_name: String, data: Variant) -> void:
	match event_name:
		"matchmaker_matched":
			_handle_matchmaker_matched(data)
		"match_state":
			# Pass match state to GameService as it handles the logic
			if "GameService" in Global:
				Global.GameService._on_match_state(data)
		"match_presence":
			# Pass match presence to GameService
			if "GameService" in Global:
				Global.GameService._on_match_presence(data)

## Creates a new match.
func create_match() -> void:
	if not core or not core.socket:
		match_error.emit("No socket connection available")
		return
		
	print("[MatchService] Creating match...")
	var result = await core.socket.create_match_async()
	
	if result is NakamaAsyncResult and result.is_exception():
		match_error.emit("Failed to create match: " + result.exception.message)
		return
		
	match_id = result.match_id
	match_created.emit(match_id)

## Joins an existing match by ID.
func join_match(m_id: String, metadata: Dictionary = {}) -> bool:
	if not core or not core.socket:
		match_error.emit("No socket connection available")
		return false
		
	print("[MatchService] Joining match: %s" % m_id)
	var result = await core.socket.join_match_async(m_id, metadata)
	
	if result is NakamaAsyncResult and result.is_exception():
		match_error.emit("Unable to join match: " + result.exception.message)
		return false
		
	match_id = result.match_id
	match_joined.emit(match_id)
	return true

## Joins the matchmaker pool.
func start_matchmaking(config: Dictionary = {}) -> bool:
	if not core or not core.socket:
		match_error.emit("No socket connection available")
		return false
		
	var min_c = config.get("min_count", min_players)
	var max_c = config.get("max_count", max_players)
	var query = config.get("query", "*")
	
	# Add client version to query if specified
	if client_version != "":
		var version_query = "+properties.client_version:" + client_version
		query = query + " " + version_query
		
	var string_props = config.get("string_properties", {})
	string_props["client_version"] = client_version
	
	print("[MatchService] Joining matchmaking pool...")
	var result = await core.socket.add_matchmaker_async(query, min_c, max_c, string_props, config.get("numeric_properties", {}))
	
	if result is NakamaAsyncResult and result.is_exception():
		match_error.emit("Unable to join matchmaking pool: " + result.exception.message)
		return false
		
	matchmaker_ticket = result.ticket
	print("[MatchService] Joined pool. Ticket: %s" % matchmaker_ticket)
	return true

func _handle_matchmaker_matched(data: NakamaRTAPI.MatchmakerMatched) -> void:
	if data is NakamaAsyncResult and data.is_exception():
		match_error.emit("Matchmaker error: " + data.exception.message)
		return
		
	print("[MatchService] Matchmaker found a match!")
	# Join the matched match
	var result = await core.socket.join_matched_async(data)
	if result is NakamaAsyncResult and result.is_exception():
		match_error.emit("Unable to join matched game: " + result.exception.message)
		return
		
	match_id = result.match_id
	match_joined.emit(match_id)

## Leaves the current match or matchmaker pool.
func leave() -> void:
	if not core or not core.socket:
		return
		
	if match_id != "":
		await core.socket.leave_match_async(match_id)
		print("[MatchService] Left match: %s" % match_id)
		match_id = ""
		
	if matchmaker_ticket != "":
		await core.socket.remove_matchmaker_async(matchmaker_ticket)
		print("[MatchService] Removed matchmaker ticket: %s" % matchmaker_ticket)
		matchmaker_ticket = ""
	
	match_disconnected.emit()

## Fetches a list of matches based on the provided filter.
func get_match_list(filter: Dictionary = {}) -> Variant:
	if not core or not core.session:
		return null
	
	var default_filter = {
		"limit": 100,
		"authoritative": true,
		"minSize": 0,
		"maxSize": 100,
		"query": "*"
	}
	if not filter.is_empty():
		for key in filter:
			default_filter[key] = filter[key]
	
	print("[MatchService] Fetching match list...")
	var result = await core.client.rpc_async(core.session, "match_list", JSON.stringify(default_filter))
	return result
