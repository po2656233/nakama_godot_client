# GameService.gd
# Handles game-specific state, player management, and the custom RPC system.
extends Node

# Dependency on NakamaCore (assumed to be in Global)
var core: Node

# Game State
var match_id: String = ""
var my_peer_id: int = 0
var players: Dictionary = {} # session_id -> Player
var _next_peer_id: int = 1

# The current active game logic instance (must extend BaseGame)
var current_game: Node = null

# Signals
signal player_joined(player: Player)
signal player_left(player: Player)
signal game_state_changed(state: int)

# Player Data Class
class Player:
	var session_id: String
	var peer_id: int
	var username: String

	func _init(_session_id: String, _username: String, _peer_id: int) -> void:
		session_id = _session_id
		username = _username
		peer_id = _peer_id

	static func from_presence(presence: NakamaRTAPI.UserPresence, _peer_id: int) -> Player:
		return Player.new(presence.session_id, presence.username, _peer_id)

	func to_dict() -> Dictionary:
		return {"session_id": session_id, "username": username, "peer_id": peer_id}

func _ready() -> void:
	if "NakamaCore" in Global:
		core = Global.NakamaCore
	else:
		push_error("[GameService] NakamaCore not found in Global!")

## Sets the active game logic instance.
func set_game(game_instance: Node) -> void:
	current_game = game_instance
	if current_game.has_method("on_game_service_attached"):
		current_game.on_game_service_attached(self)

## Initialize game session for the local player.
func init_session(m_id: String, self_user: NakamaRTAPI.UserPresence) -> void:
	match_id = m_id
	
	# Assign local player peer ID (usually 1 if creator)
	if players.is_empty():
		my_peer_id = 1
		_next_peer_id = 2
	else:
		# Find an available peer ID or use session-based order
		var session_ids = players.keys()
		session_ids.sort()
		my_peer_id = 0 # Will be set by the host via JOIN_SUCCESS
	
	var me = Player.new(self_user.session_id, self_user.username, my_peer_id)
	players[self_user.session_id] = me
	player_joined.emit(me)

## Handles match presence events (Joins/Leaves).
func _on_match_presence(data: NakamaRTAPI.MatchPresenceEvent) -> void:
	if not core or not core.socket: return
	
	for u in data.joins:
		if u.session_id == core.session.user_id: continue
		
		# Host manages peer IDs
		if is_server():
			if players.size() < 100:
				var new_player = Player.new(u.session_id, u.username, _next_peer_id)
				_next_peer_id += 1
				players[u.session_id] = new_player
				
				# Notify everyone about the new peer mapping
				send_match_state_rpc(9002, {
					"players": _serialize_players(),
					"client_version": Global.MatchService.client_version
				})
				player_joined.emit(new_player)
		else:
			pass

	for u in data.leaves:
		if u.session_id == core.session.user_id: continue
		if not players.has(u.session_id): continue
		
		var p = players[u.session_id]
		players.erase(u.session_id)
		player_left.emit(p)
		
		if p.peer_id == 1:
			match_error("Host disconnected")

## Handles match state messages.
func _on_match_state(data: NakamaRTAPI.MatchData) -> void:
	var json = JSON.new()
	if json.parse(data.data) != OK: return
	var content = json.get_data()
	
	match data.op_code:
		9001: # CUSTOM_RPC
			_handle_custom_rpc(content)
		9002: # JOIN_SUCCESS
			_handle_join_success(content)
		9003: # JOIN_ERROR
			_handle_join_error(content)

func _handle_custom_rpc(content: Dictionary) -> void:
	var peer_id = content.get("peer_id", 0)
	if peer_id != 0 and peer_id != my_peer_id: return
	
	var node_path = content.get("node_path", "")
	var method = content.get("method", "")
	var args_str = content.get("args", "[]")
	var args = str_to_var(args_str)
	
	var node = get_node(node_path)
	if not is_instance_valid(node):
		push_warning("[GameService] RPC target node not found: %s" % node_path)
		return
	
	if node.has_method(method):
		node.callv(method, args)
	else:
		push_error("[GameService] Node %s does not have method %s" % [node_path, method])

func _handle_join_success(content: Dictionary) -> void:
	var players_data = content.get("players", {})
	for session_id in players_data:
		var d = players_data[session_id]
		players[session_id] = Player.new(session_id, d.username, d.peer_id)
		if session_id == core.session.user_id:
			my_peer_id = d.peer_id
	
	print("[GameService] Joined match. My PeerID: %d" % my_peer_id)

func _handle_join_error(content: Dictionary) -> void:
	if content.get("target") == core.session.user_id:
		match_error(content.get("reason", "Unknown error"))

## Sends a custom RPC to a specific peer or everyone.
func send_rpc(node: Node, method: String, args: Array = [], target_peer_id: int = 0) -> void:
	if not core or not core.socket: return
	
	var payload = {
		"peer_id": target_peer_id,
		"node_path": node.get_path(),
		"method": method,
		"args": var_to_str(args)
	}
	
	core.socket.send_match_state_async(match_id, 9001, JSON.stringify(payload))

func send_match_state_rpc(op_code: int, data: Variant) -> void:
	if not core or not core.socket: return
	var payload = data if data is String else JSON.stringify(data)
	core.socket.send_match_state_async(match_id, op_code, payload)

# --- Helpers ---

func is_server() -> bool:
	return my_peer_id == 1

func get_peer_name(peer_id: int) -> String:
	for p in players.values():
		if p.peer_id == peer_id:
			return p.username
	return "Unknown"

func _serialize_players() -> Dictionary:
	var result = {}
	for sid in players:
		result[sid] = players[sid].to_dict()
	return result

func match_error(msg: String) -> void:
	if "WindowPopup" in Global:
		Global.WindowPopup.show_dynamic(msg)
