# NakamaCore.gd
# Core manager for the Nakama SDK. Handles client, session, and socket lifecycles.
# This service acts as the central transport layer for all other Nakama services.
extends Node

# Configuration - These can be moved to a config file or Resource
@export var server_key: String = "superman"
@export var host: String = "172.18.4.172"
@export var port: int = 8850
@export var scheme: String = "http"

var client: NakamaClient
var session: NakamaSession
var socket: NakamaSocket

# State signals
signal session_changed(new_session: NakamaSession)
signal connection_state_changed(is_connected: bool)

# Socket Event signals - These are dispatched to other services
signal socket_event_received(event_name: String, data: Variant)

# Heartbeat timer
var _heartbeat_timer: Timer

func _ready() -> void:
	_configure_network()
	_init_client()

func _configure_network() -> void:
	# Auto-configure based on platform
	if OS.has_feature("web"):
		port = 7350
		scheme = "http"
	else:
		port = 8850
		scheme = "https"

func _init_client() -> void:
	client = Nakama.create_client(
		server_key,
		host,
		port,
		scheme,
		Nakama.DEFAULT_TIMEOUT,
		NakamaLogger.LOG_LEVEL.ERROR
	)
	print("[NakamaCore] Client initialized.")

## Sets the current session and manages the socket connection accordingly.
func set_session(new_session: NakamaSession) -> void:
	session = new_session
	session_changed.emit(session)
	
	if session:
		_connect_socket()
	elif socket:
		_disconnect_socket()

func _connect_socket() -> void:
	if socket:
		socket.close()
	
	socket = Nakama.create_socket_from(client)
	_setup_socket_listeners()
	
	print("[NakamaCore] Connecting socket...")
	var result = await socket.connect_async(session, true, -1, _handle_session_reset)
	
	if result is NakamaAsyncResult and result.is_exception():
		push_error("[NakamaCore] Socket connection failed: %s" % result.exception.message)
		connection_state_changed.emit(false)
	else:
		print("[NakamaCore] Socket connected successfully.")
		connection_state_changed.emit(true)
		_start_heartbeat()

func _disconnect_socket() -> void:
	if socket:
		socket.close()
		socket = null
	connection_state_changed.emit(false)
	_stop_heartbeat()

func _handle_session_reset() -> void:
	if not session:
		return
	
	print("[NakamaCore] Attempting session reset...")
	var result = await client.session_refresh_async(session)
	if result and not result.is_exception():
		session = result
		session_changed.emit(session)
		print("[NakamaCore] Session reset successful.")
	else:
		push_error("[NakamaCore] Session reset failed.")

func _setup_socket_listeners() -> void:
	# Direct connections for core state
	socket.closed.connect(_on_socket_closed)
	socket.connected.connect(_on_socket_connected)
	socket.connection_error.connect(_on_socket_error)
	
	# Event dispatching for other services
	socket.received_matchmaker_matched.connect(func(data): socket_event_received.emit("matchmaker_matched", data))
	socket.received_match_state.connect(func(data): socket_event_received.emit("match_state", data))
	socket.received_match_presence.connect(func(data): socket_event_received.emit("match_presence", data))
	socket.received_status_presence.connect(func(data): socket_event_received.emit("status_presence", data))
	socket.received_notification.connect(func(data): socket_event_received.emit("notification", data))
	socket.received_channel_message.connect(func(data): socket_event_received.emit("channel_message", data))

func _on_socket_connected() -> void:
	connection_state_changed.emit(true)

func _on_socket_closed() -> void:
	connection_state_changed.emit(false)
	print("[NakamaCore] Socket closed.")

func _on_socket_error(error) -> void:
	push_error("[NakamaCore] Socket error: %s" % str(error))
	connection_state_changed.emit(false)

# --- Heartbeat Implementation ---

func _start_heartbeat() -> void:
	if _heartbeat_timer:
		return
	
	_heartbeat_timer = Timer.new()
	add_child(_heartbeat_timer)
	_heartbeat_timer.wait_time = 5.0
	_heartbeat_timer.timeout.connect(_send_heartbeat)
	_heartbeat_timer.start()
	print("[NakamaCore] Heartbeat started.")

func _stop_heartbeat() -> void:
	if _heartbeat_timer:
		_heartbeat_timer.stop()
		_heartbeat_timer.queue_free()
		_heartbeat_timer = null
		print("[NakamaCore] Heartbeat stopped.")

func _send_heartbeat() -> void:
	if not socket or not is_instance_valid(socket):
		return
	
	var result = await socket.rpc_async("ping")
	if result and result.is_exception():
		push_warning("[NakamaCore] Heartbeat ping failed: %s" % result.exception.message)

## Generic helper to handle Nakama async requests.
func request(callable: Callable, args: Array = []) -> Variant:
	var result = await callable.callv(args)
	if result is NakamaAsyncResult and result.is_exception():
		return result
	return result
