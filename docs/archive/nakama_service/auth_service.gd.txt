# AuthService.gd
# Handles authentication, registration, and credential management.
extends Node

# Dependency on NakamaCore (assumed to be in Global)
var core: Node

# State signals
signal auth_state_changed(is_authenticated: bool)
signal login_failed(error: String)

const CREDENTIALS_FILE = "user://credentials.json"

var current_email: String = ""
var current_password: String = ""

func _ready() -> void:
	# Ensure NakamaCore is available in Global
	if "NakamaCore" in Global:
		core = Global.NakamaCore
	else:
		push_error("[AuthService] NakamaCore not found in Global!")
	
	_load_credentials()

func _load_credentials() -> void:
	if FileAccess.file_exists(CREDENTIALS_FILE):
		var file = FileAccess.open(CREDENTIALS_FILE, FileAccess.READ)
		if file:
			var json = JSON.new()
			if json.parse(file.get_as_text()) == OK:
				var data = json.get_data()
				current_email = data.get("email", "")
				current_password = data.get("password", "")
			file.close()

func _save_credentials(email: String, password: String) -> void:
	var file = FileAccess.open(CREDENTIALS_FILE, FileAccess.WRITE)
	if file:
		var data = {"email": email, "password": password}
		file.store_line(JSON.stringify(data))
		file.close()
		print("[AuthService] Credentials saved.")

## Authenticates using email and password.
func authenticate_email(email: String, password: String, save: bool = false) -> bool:
	if email.is_empty() or password.is_empty():
		login_failed.emit("账号或密码不能为空")
		return false
		
	current_email = email.strip_edges()
	current_password = password.strip_edges()
	
	# Standardize email
	if not current_email.contains("@"):
		current_email += "@game.com"
		
	var username = current_email.split("@")[0]
	
	print("[AuthService] Authenticating email: %s" % current_email)
	var result = await core.client.authenticate_email_async(current_email, current_password, username, false)
	
	if result is NakamaAsyncResult and result.is_exception():
		var error_msg = _map_error(result.exception.message)
		login_failed.emit(error_msg)
		return false
	
	# Success
	core.set_session(result)
	if save:
		_save_ credentials(current_email, current_password)
	
	auth_state_changed.emit(true)
	return true

## Authenticates using the device's unique ID.
func authenticate_device(username: String = "", created: bool = false) -> bool:
	var device_id = OS.get_unique_id()
	var p_username = username if username != "" else null
	
	print("[AuthService] Authenticating device: %s" % device_id)
	var result = await core.client.authenticate_device_async(device_id, p_username, created)
	
	if result is NakamaAsyncResult and result.is_exception():
		var error_msg = _map_error(result.exception.message)
		login_failed.emit(error_msg)
		return false
	
	core.set_session(result)
	auth_state_changed.emit(true)
	return true

## Registers a new account using email and password.
func register(email: String, password: String) -> bool:
	if email.is_empty() or password.is_empty():
		login_failed.emit("账号或密码不能为空")
		return false
		
	var username = email.split("@")[0]
	print("[AuthService] Registering account: %s" % email)
	
	var result = await core.client.authenticate_email_async(email, password, username, true)
	
	if result is NakamaAsyncResult and result.is_exception():
		var error_msg = _map_error(result.exception.message)
		login_failed.emit(error_msg)
		return false
	
	core.set_session(result)
	auth_state_changed.emit(true)
	return true

## Logs out the current user.
func logout(clear_cache: bool = true) -> void:
	if core.session:
		await core.client.session_logout_async(core.session)
	
	core.set_session(null)
	auth_state_changed.emit(false)
	
	if clear_cache:
		if FileAccess.file_exists(CREDENTIALS_FILE):
			DirAccess.remove_absolute(CREDENTIALS_FILE)
			print("[AuthService] Cache cleared.")
	
	print("[AuthService] User logged out.")

func relogin() -> void:
	if current_email == "" or current_password == "":
		push_warning("[AuthService] No credentials available for relogin.")
		return
	await authenticate_email(current_email, current_password)

func _map_error(message: String) -> String:
	match message:
		"Invalid credentials.": return "账号或密码错误"
		"User account not found.": return "该账号未注册"
		"Password must be at least 8 characters long.": return "密码长度至少为8位"
		"HTTPRequest failed!": return "网络连接失败，请检查您的网络设置"
		"": return "登录失败，请稍后重试"
	return message
