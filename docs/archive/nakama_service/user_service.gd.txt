# UserService.gd
# Handles retrieval and persistence of user-specific data.
extends Node

# Dependency on NakamaCore (assumed to be in Global)
var core: Node

# Cached user profile
var current_user: NakamaAPI.ApiUser

# State signals
signal user_data_updated(user: NakamaAPI.ApiUser)

func _ready() -> void:
	if "NakamaCore" in Global:
		core = Global.NakamaCore
	else:
		push_error("[UserService] NakamaCore not found in Global!")

## Fetches the current authenticated user's profile from Nakama.
func load_user_data() -> void:
	if not core or not core.session:
		push_warning("[UserService] Cannot load user data: No active session.")
		return
	
	print("[UserService] Fetching user data for: %s" % core.session.user_id)
	var result = await core.client.get_users_async(core.session, [core.session.user_id])
	
	if result is NakamaAsyncResult and result.is_exception():
		push_error("[UserService] Failed to load user data: %s" % result.exception.message)
		return
	
	if result.users and result.users.size() > 0:
		current_user = result.users[0]
		user_data_updated.emit(current_user)
		print("[UserService] User data updated: %s" % current_user.username)
	else:
		push_error("[UserService] No user found for session ID: %s" % core.session.user_id)

## Persists a dictionary of profile data to Nakama storage.
func save_user_profile(profile_data: Dictionary) -> bool:
	if not core or not core.session:
		push_warning("[UserService] Cannot save profile: No active session.")
		return false
		
	var objects = [{
		"collection": "player_data",
		"key": "profile",
		"value": JSON.stringify(profile_data),
		"permission_read": 1,
		"permission_write": 1
	}]
	
	var result = await core.client.write_storage_objects_async(core.session, objects)
	
	if result is NakamaAsyncResult and result.is_exception():
		push_error("[UserService] Failed to save profile: %s" % result.exception.message)
		return false
		
	print("[UserService] User profile saved successfully.")
	return true

# --- Getters ---

func get_my_id() -> String:
	return current_user.id if current_user else ""

func get_my_name() -> String:
	return current_user.username if current_user else ""

func get_my_display_name() -> String:
	return current_user.display_name if current_user else ""

func get_my_avatar() -> String:
	return current_user.avatar_url if current_user else ""
